# バックエンド（別ターミナル。コード変更後は必ず再起動）
cd c:\Python\ForTeachingEveryone
pip install -r backend\requirements.txt
# ポートは 8010（8000 は Windows で予約され WinError 10013 になることが多い）
cd backend
python -m uvicorn main:app --reload --host 127.0.0.1 --port 8010

# 動作確認（ブラウザまたは curl）
# http://127.0.0.1:8010/department
# http://127.0.0.1:8010/docs
